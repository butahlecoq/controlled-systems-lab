cmake_minimum_required(VERSION 3.31)
project(CSLMPC11Affine LANGUAGES C CXX)
if(NOT DEFINED CSL_SOURCE_ROOT)
  message(FATAL_ERROR "CSL_SOURCE_ROOT must name the source checkout")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Repository tests" FORCE)
add_subdirectory("${CSL_SOURCE_ROOT}" csl-source)
include("${CSL_SOURCE_ROOT}/cmake/JsonDependencies.cmake")
FetchContent_GetProperties(csl_json SOURCE_DIR csl_mpc11_json_source)
add_executable(csl_mpc11_affine affine_bridge.cpp)
target_link_libraries(csl_mpc11_affine PRIVATE CSL::highs_convex CSL::convex)
target_include_directories(csl_mpc11_affine SYSTEM PRIVATE ${csl_mpc11_json_source}/single_include)
target_compile_features(csl_mpc11_affine PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(csl_mpc11_affine PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(csl_mpc11_affine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
