cmake_minimum_required(VERSION 3.31)
project(CSLSYN04StrategyBMeasurement LANGUAGES C CXX)

if(NOT DEFINED CSL_SOURCE_ROOT)
  message(FATAL_ERROR "CSL_SOURCE_ROOT must name the pinned clean solver checkout")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Build repository tests" FORCE)
add_subdirectory("${CSL_SOURCE_ROOT}" csl-source)

add_executable(csl_syn04_strategy_b_report "${CMAKE_CURRENT_LIST_DIR}/../strategy_b_measure.cpp")
target_link_libraries(csl_syn04_strategy_b_report PRIVATE CSL::strategy_b CSL::highs_lp CSL::highs_convex)
target_compile_features(csl_syn04_strategy_b_report PRIVATE cxx_std_20)
target_compile_definitions(csl_syn04_strategy_b_report PRIVATE $<$<CONFIG:Release>:NDEBUG>)
if(MSVC)
  target_compile_options(csl_syn04_strategy_b_report PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(csl_syn04_strategy_b_report PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
