cmake_minimum_required(VERSION 3.31)
project(CSLMPC10 LANGUAGES C CXX)
foreach(required CSL_SOURCE_ROOT CSL_EXISTING_HIGHS_LIBRARY CSL_EXISTING_HIGHS_SOURCE CSL_EXISTING_HIGHS_BINARY CSL_EXISTING_JSON_SOURCE)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "${required} must identify the recorded existing dependency")
  endif()
endforeach()
set(BUILD_TESTING OFF CACHE BOOL "Tests" FORCE)
set(CSL_ENABLE_HIGHS OFF CACHE BOOL "Reuse existing native archive" FORCE)
add_subdirectory("${CSL_SOURCE_ROOT}" csl-source)
add_library(mpc10_native_highs STATIC IMPORTED)
set_target_properties(mpc10_native_highs PROPERTIES IMPORTED_LOCATION "${CSL_EXISTING_HIGHS_LIBRARY}")
add_library(mpc10_highs_adapters STATIC "${CSL_SOURCE_ROOT}/src/adapters/lp/highs.cpp" "${CSL_SOURCE_ROOT}/src/adapters/convex/highs.cpp")
target_link_libraries(mpc10_highs_adapters PUBLIC CSL::optimization CSL::convex CSL::registry PRIVATE mpc10_native_highs)
target_include_directories(mpc10_highs_adapters SYSTEM PRIVATE "${CSL_EXISTING_HIGHS_SOURCE}" "${CSL_EXISTING_HIGHS_SOURCE}/highs" "${CSL_EXISTING_HIGHS_BINARY}")
target_compile_features(mpc10_highs_adapters PRIVATE cxx_std_20)
target_compile_definitions(mpc10_highs_adapters PRIVATE
  CSL_HIGHS_PATCHED_SOURCE_SHA256="8ee92e023806826fded545e38d0598adacc7b08f07c395c557a1dbaa1733b00b"
  CSL_HIGHS_THREAD_OBSERVATION_PATCH_SHA256="5e81b644fad40522a62037b0dab8ae0f023a8f7db7540297030ac28de58b56cc"
  CSL_HIGHS_CLOCK_SNAPSHOT_PATCH_SHA256="9e6849f8933e3e434c5964f78f2c7229537a9398f1d6ded35a2cd2c5d1839501")
add_executable(csl_mpc10 bridge.cpp)
target_link_libraries(csl_mpc10 PRIVATE CSL::hybrid_path_program CSL::strategy_b CSL::finite_mixed_branch mpc10_highs_adapters)
target_include_directories(csl_mpc10 SYSTEM PRIVATE "${CSL_EXISTING_JSON_SOURCE}/single_include")
target_compile_features(csl_mpc10 PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(csl_mpc10 PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(csl_mpc10 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
