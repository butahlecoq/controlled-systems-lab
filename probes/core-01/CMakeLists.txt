cmake_minimum_required(VERSION 3.31)
project(csl_core01_probes LANGUAGES C CXX)
include(FetchContent)

# Isolated integration project; CORE-02 owns the future product build.
set(BUILD_TESTING OFF CACHE BOOL "Disable dependency tests" FORCE)
set(BUILD_CXX_EXE OFF CACHE BOOL "Library only" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "Disable dependency examples" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static probe dependencies" FORCE)
set(ZLIB OFF CACHE BOOL "No compressed solver files" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "Disable dependency tests" FORCE)
set(DOCTEST_WITH_TESTS OFF CACHE BOOL "Disable dependency tests" FORCE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
FetchContent_Declare(doctest URL https://codeload.github.com/doctest/doctest/tar.gz/refs/tags/v2.4.11
  URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186)
FetchContent_MakeAvailable(doctest)
enable_testing()

add_library(csl_contracts STATIC ../../src/contracts.cpp)
target_include_directories(csl_contracts PUBLIC ../../include)
target_compile_features(csl_contracts PUBLIC cxx_std_20)
set_target_properties(csl_contracts PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(csl_contracts PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(csl_contracts PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(contract_tests contract_tests.cpp)
target_link_libraries(contract_tests PRIVATE csl_contracts doctest::doctest)
add_test(NAME contract_tests COMMAND contract_tests)
find_package(Threads REQUIRED)
add_executable(caller_example caller_example_contracts.cpp)
target_link_libraries(caller_example PRIVATE csl_contracts Threads::Threads)
add_test(NAME caller_example COMMAND caller_example)

option(CSL_DEPENDENCY_PROBES "Build numerical and parser probes" ON)
if(CSL_DEPENDENCY_PROBES)
  # Eigen is consumed as headers, avoiding its own test/install build graph.
  FetchContent_Declare(eigen URL https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz
    URL_HASH SHA256=8586084f71f9bde545ee7fa6d00288b264a2b7ac3607b974e54d13e7162c1c72
    SOURCE_SUBDIR csl-no-build)
  FetchContent_Declare(highs URL https://codeload.github.com/ERGO-Code/HiGHS/tar.gz/refs/tags/v1.11.0
    URL_HASH SHA256=2b44b074cf41439325ce4d0bbdac2d51379f56faf17ba15320a410d3c1f07275)
  FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3
    URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
  FetchContent_MakeAvailable(eigen highs json)
  add_executable(dependency_probes dependency_probes.cpp)
  target_compile_features(dependency_probes PRIVATE cxx_std_20)
  target_compile_definitions(dependency_probes PRIVATE EIGEN_MPL2_ONLY)
  target_include_directories(dependency_probes SYSTEM PRIVATE ${eigen_SOURCE_DIR})
  target_link_libraries(dependency_probes PRIVATE highs nlohmann_json::nlohmann_json doctest::doctest)
  add_test(NAME dependency_probes COMMAND dependency_probes)
endif()
