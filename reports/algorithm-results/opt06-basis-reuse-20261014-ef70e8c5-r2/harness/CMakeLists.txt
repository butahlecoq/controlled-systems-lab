cmake_minimum_required(VERSION 3.31)
project(CSLOPT06ReportHarness LANGUAGES C CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CSL_BUILD_REPORTS OFF CACHE BOOL "" FORCE)
if(NOT DEFINED CSL_ROOT)
  get_filename_component(CSL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../" ABSOLUTE)
endif()
add_subdirectory("${CSL_ROOT}" csl)
add_executable(csl_opt06_measure opt06_measure.cpp)
target_link_libraries(csl_opt06_measure PRIVATE CSL::hull CSL::highs_lp)
target_compile_features(csl_opt06_measure PRIVATE cxx_std_20)
set_target_properties(csl_opt06_measure PROPERTIES CXX_EXTENSIONS OFF)
