cmake_minimum_required(VERSION 3.31)
project(CSLMpc06Online LANGUAGES C CXX)
if(NOT DEFINED CSL_SOURCE_ROOT)
  message(FATAL_ERROR "CSL_SOURCE_ROOT is required")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Tests are run by the parent project" FORCE)
add_subdirectory("${CSL_SOURCE_ROOT}" csl-source)
add_executable(csl_mpc06_online driver.cpp)
target_link_libraries(csl_mpc06_online PRIVATE CSL::highs_convex CSL::approximation)
target_compile_features(csl_mpc06_online PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(csl_mpc06_online PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(csl_mpc06_online PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
