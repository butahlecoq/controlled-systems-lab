cmake_minimum_required(VERSION 3.31)
project(csl_row_scaling_diagnostic LANGUAGES CXX)
find_package(CSL CONFIG REQUIRED)
add_executable(row_scaling_driver row_scaling_driver.cpp)
target_compile_features(row_scaling_driver PRIVATE cxx_std_20)
target_compile_options(row_scaling_driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(row_scaling_driver PRIVATE CSL::model_catalogue CSL::nominal_quadratic_history CSL::highs_convex)
add_executable(analytic analytic.cpp)
target_compile_features(analytic PRIVATE cxx_std_20)
target_compile_options(analytic PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(analytic PRIVATE CSL::highs_convex)
