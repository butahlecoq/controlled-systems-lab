cmake_minimum_required(VERSION 3.31)
project(CSL VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(CSL_DEPENDENCY_PROBES "Build the isolated pinned native dependency probes" OFF)

add_library(csl_contracts STATIC src/contracts.cpp)
add_library(CSL::contracts ALIAS csl_contracts)
set_target_properties(csl_contracts PROPERTIES EXPORT_NAME contracts CXX_EXTENSIONS OFF)
add_library(csl_numeric STATIC src/numeric/data.cpp)
add_library(CSL::numeric ALIAS csl_numeric)
set_target_properties(csl_numeric PROPERTIES EXPORT_NAME numeric CXX_EXTENSIONS OFF)
target_link_libraries(csl_numeric PUBLIC CSL::contracts)
add_library(csl_objectives STATIC src/optimization/objective_reformulation.cpp)
add_library(CSL::objectives ALIAS csl_objectives)
set_target_properties(csl_objectives PROPERTIES EXPORT_NAME objectives CXX_EXTENSIONS OFF)
target_link_libraries(csl_objectives PUBLIC CSL::numeric)
include(cmake/ObjectivesDependencies.cmake)
add_library(csl_registry STATIC src/registry/registry.cpp)
add_library(CSL::registry ALIAS csl_registry)
set_target_properties(csl_registry PROPERTIES EXPORT_NAME registry CXX_EXTENSIONS OFF)
target_link_libraries(csl_registry PUBLIC CSL::contracts)
add_library(csl_sets STATIC src/sets/data.cpp)
add_library(CSL::sets ALIAS csl_sets)
set_target_properties(csl_sets PROPERTIES EXPORT_NAME sets CXX_EXTENSIONS OFF)
target_link_libraries(csl_sets PUBLIC CSL::numeric)
add_library(csl_model STATIC src/model/linear_system.cpp)
add_library(CSL::model ALIAS csl_model)
set_target_properties(csl_model PROPERTIES EXPORT_NAME model CXX_EXTENSIONS OFF)
target_link_libraries(csl_model PUBLIC CSL::numeric)
add_library(csl_optimization STATIC src/optimization/linear_program.cpp src/optimization/lp_backend.cpp)
add_library(CSL::optimization ALIAS csl_optimization)
set_target_properties(csl_optimization PROPERTIES EXPORT_NAME optimization CXX_EXTENSIONS OFF)
target_link_libraries(csl_optimization PUBLIC CSL::numeric)
foreach(target IN ITEMS csl_contracts csl_numeric csl_registry csl_sets csl_model csl_objectives csl_optimization)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()

install(TARGETS csl_contracts csl_numeric csl_registry csl_sets csl_model csl_objectives csl_optimization EXPORT CSLTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/csl/contracts.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl)
install(FILES include/csl/numeric/data.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/numeric)
install(FILES include/csl/optimization/objectives.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/optimization)
install(FILES include/csl/registry/registry.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/registry)
install(FILES include/csl/sets/data.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/sets)
install(FILES include/csl/model/linear_system.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/model)
install(FILES include/csl/optimization/linear_program.hpp include/csl/optimization/lp_backend.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csl/optimization)
install(EXPORT CSLTargets NAMESPACE CSL:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CSL)
configure_package_config_file(cmake/CSLConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/CSLConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CSL)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/CSLConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/CSLConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/CSLConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CSL)
