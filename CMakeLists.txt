if(CSL_ENABLE_QHULL AND NOT CSL_ENABLE_HIGHS)
  message(FATAL_ERROR "CSL_ENABLE_QHULL requires CSL_ENABLE_HIGHS for its default LP provider")
endif()
if(CSL_DEPENDENCY_PROBES AND (NOT BUILD_TESTING OR NOT CSL_ENABLE_HIGHS OR NOT CSL_ENABLE_QHULL))
  message(FATAL_ERROR "Dependency probes require tests and both native adapters")
endif()
find_package(Threads REQUIRED)
include(cmake/ModulePolicy.cmake)

# Include module-owned declarations in the project scope so dependency pins and
# package state remain shared. The same order is used for focused test ownership.
set(csl_modules contracts numeric optimization registry sets model controllers geometry problem
  reachability synthesis verification input execution artifacts adapters/highs adapters/polyhedral application)
foreach(module IN LISTS csl_modules)
  include(src/${module}/Module.cmake)
endforeach()
include(cmake/OptionalReports.cmake)
include(cmake/Package.cmake)
if(BUILD_TESTING)
  include(cmake/Dependencies.cmake)
  include(cmake/TestCatalog.cmake)
  foreach(module IN LISTS csl_modules)
    include(src/${module}/Tests.cmake)
  endforeach()
  include(cmake/InstalledTests.cmake)
endif()
if(CSL_ENABLE_HIGHS)
  get_property(csl_tests_before DIRECTORY PROPERTY TESTS)
  include(cmake/ReportTargets.cmake)
  if(BUILD_TESTING)
    get_property(csl_report_tests DIRECTORY PROPERTY TESTS)
    list(REMOVE_ITEM csl_report_tests ${csl_tests_before})
    set_tests_properties(${csl_report_tests} PROPERTIES LABELS reporting)
    add_custom_target(csl_reporting_checks DEPENDS lp_report_tests convex_report_tests csl_lp_report csl_convex_report)
  endif()
endif()
