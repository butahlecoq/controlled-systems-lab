cmake_minimum_required(VERSION 3.31)
project(SYN04Slice LANGUAGES C CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
get_filename_component(CSL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${CSL_ROOT}" csl)
add_library(syn04_slice STATIC "${CSL_ROOT}/src/synthesis/strategy_b/direct_joint.cpp")
target_link_libraries(syn04_slice PUBLIC CSL::resolved_problem CSL::sets CSL::registry CSL::convex CSL::verification CSL::artifacts)
target_compile_features(syn04_slice PUBLIC cxx_std_20)
add_executable(syn04_tests direct_joint_tests.cpp)
target_include_directories(syn04_tests PRIVATE "${CSL_ROOT}/src/registry")
target_link_libraries(syn04_tests PRIVATE syn04_slice doctest::doctest CSL::highs_lp CSL::highs_convex
  CSL::strategy_a CSL::affine_sum CSL::qhull_conversion)
if(MSVC)
  target_compile_options(syn04_slice PRIVATE /W4 /WX /permissive-)
  target_compile_options(syn04_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(syn04_slice PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(syn04_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME syn04_tests COMMAND syn04_tests)
